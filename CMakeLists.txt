cmake_minimum_required(VERSION 3.16)
project(isekai LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release)
endif()
option(ISEKAI_WINDOW "Build interactive OpenGL window mode (GLFW)" OFF)

add_executable(isekai src/main.cpp src/gltf_loader.cpp src/soft_raster.cpp)
target_include_directories(isekai PRIVATE include third_party)

if(ISEKAI_WINDOW)
  include(FetchContent)
  FetchContent_Declare(glfw GIT_REPOSITORY https://github.com/glfw/glfw.git GIT_TAG 3.4)
  set(GLFW_BUILD_DOCS OFF CACHE BOOL "" FORCE)
  set(GLFW_BUILD_TESTS OFF CACHE BOOL "" FORCE)
  set(GLFW_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
  FetchContent_MakeAvailable(glfw)
  target_link_libraries(isekai PRIVATE glfw)
  target_compile_definitions(isekai PRIVATE ISEKAI_WINDOW=1)
endif()

if(MSVC)
  target_compile_options(isekai PRIVATE /W3 /O2)
else()
  target_compile_options(isekai PRIVATE -Wall -Wextra -O2)
endif()
